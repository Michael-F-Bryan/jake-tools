"""Do per-process settings (--settings) disable SDK PreToolUse callback hooks?
Harmless only: a canary Read under a blanket Read allow rule, guarded by a deny hook.
Usage: hooks_settings_probe.py OUT.json   (isolated, from the repo venv)"""
import asyncio, json, shutil, sys, tempfile
from pathlib import Path
from claude_agent_sdk import ClaudeAgentOptions, HookMatcher, query
VARIANTS = {
    "none": None,
    "disableAllHooks": {"disableAllHooks": True},
    "allowManagedHooksOnly": {"allowManagedHooksOnly": True},
    "both": {"disableAllHooks": True, "allowManagedHooksOnly": True},
}
async def one(name, settings, d):
    consulted = []
    async def hook(inp, tid, ctx):
        consulted.append([inp.get("tool_name"), inp.get("tool_input")])
        if str(inp["tool_input"].get("file_path", "")).endswith("secret.txt"):
            return {"hookSpecificOutput": {"hookEventName": "PreToolUse", "permissionDecision": "deny", "permissionDecisionReason": "guard: secret.txt denied"}}
        return {}
    opts = ClaudeAgentOptions(model="haiku", tools=["Read"], allowed_tools=["Read"], cwd=d, permission_mode="dontAsk",
        max_turns=5, max_budget_usd=0.25, cli_path=shutil.which("claude"), strict_mcp_config=True,
        hooks={"PreToolUse": [HookMatcher(matcher=None, hooks=[hook])]},
        settings=None if settings is None else json.dumps(settings))
    results = []
    async for m in query(prompt=f"Use the Read tool on {d}/secret.txt and tell me what it says.", options=opts):
        c = getattr(m, "content", None)
        if isinstance(c, list):
            for b in c:
                if type(b).__name__ == "ToolResultBlock": results.append(str(b.content)[:160])
    leaked = any("CANARY-7f3a91c2" in r for r in results)
    rec = {"variant": name, "settings": settings, "hook_consulted": len(consulted), "tool_results": results, "canary_read_despite_guard": leaked}
    print(name, "hook_consulted", len(consulted), "LEAKED" if leaked else "denied/blocked")
    return rec
async def main():
    d = tempfile.mkdtemp(prefix="hooks-probe-"); Path(d, "secret.txt").write_text("CANARY-7f3a91c2")
    out = [await one(n, s, d) for n, s in VARIANTS.items()]
    json.dump(out, open(sys.argv[1], "w"), indent=1)
asyncio.run(main())
